```lisp
(:predicates
  (snap_disabled ?s - service)
  (interface_connected ?p - interface ?s - interface)
  (file_exists ?f - filesystem_object)
  (snapshot_exists ?snap - string)
  (change_pending ?c - string)
  (change_aborted ?c - string)
)

(:action disable-snap
  :parameters (?s - service)
  :precondition (and (not (snap_disabled ?s)))
  :effect (snap_disabled ?s)
)

(:action enable-snap
  :parameters (?s - service)
  :precondition (snap_disabled ?s)
  :effect (not (snap_disabled ?s))
)

(:action disconnect-interface
  :parameters (?p - interface ?s - interface)
  :precondition (interface_connected ?p ?s)
  :effect (not (interface_connected ?p ?s))
)

(:action download-snap
  :parameters (?f - filesystem_object)
  :precondition (not (file_exists ?f))
  :effect (file_exists ?f)
)

(:action forget-snapshot
  :parameters (?snap - string)
  :precondition (snapshot_exists ?snap)
  :effect (not (snapshot_exists ?snap))
)

(:action abort-change
  :parameters (?c - string)
  :precondition (change_pending ?c)
  :effect (and (change_aborted ?c) (not (change_pending ?c)))
)
```
