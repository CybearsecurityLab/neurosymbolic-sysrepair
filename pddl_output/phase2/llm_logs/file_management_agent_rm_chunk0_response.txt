(:predicates 
  (file_exists ?f - filesystem_object)
  (file_writable ?f - filesystem_object)
  (directory_empty ?d - directory)
)

(:action remove-file
 :parameters (?f - file)
 :precondition (and (file_exists ?f) (file_writable ?f))
 :effect (not (file_exists ?f))
)

(:action remove-file-force
 :parameters (?f - file)
 :precondition (file_exists ?f)
 :effect (not (file_exists ?f))
)

(:action remove-directory-recursive
 :parameters (?d - directory)
 :precondition (file_exists ?d)
 :effect (not (file_exists ?d))
)

(:action remove-empty-directory
 :parameters (?d - directory)
 :precondition (and (file_exists ?d) (directory_empty ?d))
 :effect (not (file_exists ?d))
)
