(:predicates 
  (file_critical ?f - filesystem_object)
  (file_writable ?f - filesystem_object)
  (file_exists ?f - filesystem_object)
  (config_applied ?f - configuration_file))

(:action save-top-configuration
 :parameters (?f - configuration_file)
 :precondition (and (file_critical ?f) (file_writable ?f))
 :effect (and (file_exists ?f) (config_applied ?f)))
