```pddl
(:types
  interface port firewall_rule - object
)

(:predicates
  (interface_exists ?i - interface)
  (interface_up ?i - interface)
  (port_allowed ?p - port)
  (port_open ?p - port)
  (firewall_rule_exists ?r - firewall_rule)
  (traffic_blocked ?r - firewall_rule)
)

(:action configure_interface
  :parameters (?i - interface)
  :precondition (and (interface_exists ?i))
  :effect (and (interface_up ?i))
)

(:action open_port
  :parameters (?p - port)
  :precondition (and (port_allowed ?p))
  :effect (and (port_open ?p))
)

(:action create_firewall_rule
  :parameters (?r - firewall_rule)
  :precondition (and (not (firewall_rule_exists ?r)))
  :effect (and (firewall_rule_exists ?r))
)

(:action block_traffic
  :parameters (?r - firewall_rule)
  :precondition (and (firewall_rule_exists ?r))
  :effect (and (traffic_blocked ?r))
)
```
