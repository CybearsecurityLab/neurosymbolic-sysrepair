(:types
  process - object
)

(:predicates
  (command_column_right_justified ?p - process)
  (command_lines_truncated ?p - process)
  (window_displayed ?p - process)
  (forest_view_active ?p - process)
  (interactive_mode_active ?p - process)
)

(:action toggle_command_column_justification
  :parameters (?p - process)
  :precondition (and (window_displayed ?p) (interactive_mode_active ?p))
  :effect (and (command_column_right_justified ?p))
)

(:action toggle_forest_view
  :parameters (?p - process)
  :precondition (and (window_displayed ?p) (interactive_mode_active ?p))
  :effect (and (forest_view_active ?p))
)

(:action resize_window
  :parameters (?p - process)
  :precondition (and (window_displayed ?p) (interactive_mode_active ?p))
  :effect (and (command_lines_truncated ?p))
)

(:action navigate_to_command_column
  :parameters (?p - process)
  :precondition (and (window_displayed ?p) (interactive_mode_active ?p) (command_column_right_justified ?p))
  :effect (and (command_lines_truncated ?p))
)
