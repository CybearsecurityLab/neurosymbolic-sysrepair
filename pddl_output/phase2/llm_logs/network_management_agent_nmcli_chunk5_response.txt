```lisp
(:predicates
  (interface_disconnected ?i - interface)
  (autoconnect_disabled ?i - interface)
  (wifi_connection_exists ?c - configuration_file)
  (ipv4_method_configured ?i - interface ?m - string)
  (connection_autoconnect_set ?c - configuration_file ?v - boolean)
)

(:action activate_connection
 :parameters (?c - configuration_file ?i - interface)
 :precondition (and (interface_exists ?i) (wifi_connection_exists ?c))
 :effect (config_applied ?c)
)

(:action deactivate_device
 :parameters (?i - interface)
 :precondition (interface_exists ?i)
 :effect (and (interface_disconnected ?i) (autoconnect_disabled ?i))
)

(:action create_wifi_connection
 :parameters (?ssid - string ?password - string ?c - configuration_file ?i - interface)
 :precondition (interface_exists ?i)
 :effect (wifi_connection_exists ?c)
)

(:action modify_device_ipv4_method
 :parameters (?i - interface ?method - string)
 :precondition (interface_exists ?i)
 :effect (ipv4_method_configured ?i ?method)
)

(:action modify_connection_autoconnect
 :parameters (?c - configuration_file ?value - boolean)
 :precondition (wifi_connection_exists ?c)
 :effect (connection_autoconnect_set ?c ?value)
)
```
