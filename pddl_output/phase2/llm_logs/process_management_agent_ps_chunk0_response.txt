(:predicates 
  (all_processes_selected)
  (non_session_leader_selected)
  (tty_processes_selected)
  (include_non_tty_processes)
  (running_processes_selected)
  (terminal_processes_selected)
  (command_name_selected ?cmd - object)
  (process_id_selected ?pid - object)
  (group_id_selected ?gid - object)
  (session_id_selected ?sid - object)
  (tty_selected ?tty - object)
)

(:action select-all-processes
  :effect (all_processes_selected)
)

(:action select-non-session-tty
  :effect (and 
    (non_session_leader_selected) 
    (tty_processes_selected)
  )
)

(:action include-non-tty
  :effect (include_non_tty_processes)
)

(:action select-running
  :effect (running_processes_selected)
)

(:action select-terminal
  :effect (terminal_processes_selected)
)

(:action select-by-command
  :parameters (?cmd - object)
  :effect (command_name_selected ?cmd)
)

(:action select-by-pid
  :parameters (?pid - object)
  :effect (process_id_selected ?pid)
)

(:action select-by-group
  :parameters (?gid - object)
  :effect (group_id_selected ?gid)
)

(:action select-by-sid
  :parameters (?sid - object)
  :effect (session_id_selected ?sid)
)

(:action select-by-tty
  :parameters (?tty - object)
  :effect (tty_selected ?tty)
)
