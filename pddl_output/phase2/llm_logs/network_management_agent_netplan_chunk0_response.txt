```lisp
(:predicates
  (configures ?f - configuration_file ?s - service)
  (file_exists ?f - filesystem_object)
  (file_readable ?f - filesystem_object)
  (file_writable ?f - filesystem_object)
  (config_applied ?s - service)
)

(:action netplan-generate
  :parameters (?yaml - configuration_file ?generated - configuration_file ?service - service)
  :precondition (and (file_exists ?yaml) (file_readable ?yaml))
  :effect (and (file_exists ?generated) (configures ?generated ?service))
)

(:action netplan-apply
  :parameters (?config - configuration_file ?service - service)
  :precondition (and (file_exists ?config) (configures ?config ?service))
  :effect (config_applied ?service)
)

(:action netplan-set
  :parameters (?yaml - configuration_file ?service - service)
  :precondition (and (file_exists ?yaml) (file_writable ?yaml))
  :effect (configures ?yaml ?service)
)
```
