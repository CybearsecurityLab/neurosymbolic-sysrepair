(:predicates
  (user_has_primary_group ?u - user ?g - group)
  (no_users_with_primary_group ?g - group)
  (group_exists ?g - group))

(:action delete-group
  :parameters (?g - group)
  :precondition (and (group_exists ?g) (no_users_with_primary_group ?g))
  :effect (not (group_exists ?g)))

(:action force-delete-group
  :parameters (?g - group)
  :precondition (group_exists ?g)
  :effect (not (group_exists ?g)))
