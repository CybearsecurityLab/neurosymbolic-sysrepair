 :types
  service-state - object
  configuration-file - object

(:predicates
  (service-running ?s - service)
  (service-enabled ?s - service)
  (service-failed ?s - service)
  (configuration-applied ?c - configuration_file ?s - service)
)

(:action start_service
  :parameters (?s - service)
  :precondition (and (service-exists ?s) (service-enabled ?s))
  :effect (and (service-running ?s)))

(:action stop_service
  :parameters (?s - service)
  :precondition (and (service-running ?s))
  :effect (and (not (service-running ?s))))

(:action enable_service
  :parameters (?s - service)
  :precondition (and (service-exists ?s))
  :effect (and (service-enabled ?s)))

(:action disable_service
  :parameters (?s - service)
  :precondition (and (service-enabled ?s))
  :effect (and (not (service-enabled ?s))))

(:action reload_service
  :parameters (?s - service)
  :precondition (and (service-running ?s))
  :effect (and (not (service-running ?s)) (service-running ?s)))

(:action apply_configuration
  :parameters (?c - configuration_file ?s - service)
  :precondition (and (file_exists ?c) (configures ?c ?s))
  :effect (and (config_applied ?c ?s)))

(:action revert_configuration
  :parameters (?c - configuration_file ?s - service)
  :precondition (and (config_applied ?c ?s))
  :effect (and (not (config_applied ?c ?s))))
