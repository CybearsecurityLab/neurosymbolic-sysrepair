 :types
  service-state - enum {enabled, disabled, failed, running}
  configuration-file - object

(:predicates
  (service-exists ?s - service)
  (service-state ?s - service service-state)
  (service-configured-file ?s - service configuration-file)
)

(:action start_service
  :parameters (?s - service)
  :precondition (and (service-exists ?s))
  :effect (and (not (service-failed ?s)) (not (service-disabled ?s)) (service-running ?s)))

(:action stop_service
  :parameters (?s - service)
  :precondition (and (service-exists ?s) (service-running ?s))
  :effect (and (not (service-running ?s)) (not (service-failed ?s))))

(:action enable_service
  :parameters (?s - service)
  :precondition (and (service-exists ?s))
  :effect (and (not (service-disabled ?s)) (not (service-failed ?s))))

(:action disable_service
  :parameters (?s - service)
  :precondition (and (service-exists ?s))
  :effect (and (service-disabled ?s)))

(:action reload_service
  :parameters (?s - service)
  :precondition (and (service-exists ?s) (service-running ?s))
  :effect (and (not (service-failed ?s))))

(:action check_service_status
  :parameters (?s - service)
  :effect (and (service-state ?s)))

(:action configure_service_file
  :parameters (?s - service ?f - configuration_file)
  :precondition (and (service-exists ?s) (file_exists ?f))
  :effect (and (service-configured-file ?f ?s)))

(:action revert_service_configuration
  :parameters (?s - service)
  :precondition (and (service-exists ?s) (service-configured-file ?f ?s))
  :effect (and (not (service-configured-file ?f ?s))))

(:action check_service_configuration
  :parameters (?s - service)
  :effect (and (service-configured-file ?f ?s)))
