```lisp
(:predicates
  (firewall_rule_exists ?r - firewall_rule)
  (config_applied ?s - service)
  (configures ?f - configuration_file ?s - service)
)

(:action add-firewall-rule
  :parameters (?r - firewall_rule)
  :precondition (not (firewall_rule_exists ?r))
  :effect (firewall_rule_exists ?r)
)

(:action reload-ufw
  :parameters (?s - service)
  :precondition (and (configures ?f - configuration_file ?s))
  :effect (config_applied ?s)
)
```
