(:predicates
  (queue_rule_exists ?r - firewall_rule)
  (set_element_exists ?s - configuration_file ?e - string)
  (map_element_exists ?m - configuration_file ?k - string ?v - string)
  (interface_configured ?i - interface)
)

(:action add_queue_rule
  :parameters (?r - firewall_rule ?q - integer ?flag - string ?expr - string)
  :precondition (and (file_critical ?r) (file_executable ?r))
  :effect (and (queue_rule_exists ?r)
               (config_applied ?r)
               (configures ?r ?q))
)

(:action duplicate_packet
  :parameters (?r - firewall_rule ?addr - string ?dev - string)
  :precondition (and (interface_exists ?dev) (file_critical ?r))
  :effect (and (firewall_rule_exists ?r)
               (config_applied ?r)
               (configures ?r ?addr)
               (configures ?r ?dev))
)

(:action forward_packet
  :parameters (?r - firewall_rule ?addr - string ?dev - string)
  :precondition (and (interface_exists ?dev) (file_critical ?r))
  :effect (and (firewall_rule_exists ?r)
               (config_applied ?r)
               (configures ?r ?addr)
               (configures ?r ?dev))
)

(:action update_set
  :parameters (?s - configuration_file ?expr - string ?timeout - integer ?comment - string)
  :precondition (and (file_critical ?s) (file_writable ?s))
  :effect (and (set_element_exists ?s ?expr)
               (config_applied ?s)
               (configures ?s ?expr))
)

(:action create_map
  :parameters (?m - configuration_file ?key - string ?value - string)
  :precondition (and (file_critical ?m) (file_writable ?m))
  :effect (and (map_element_exists ?m ?key ?value)
               (config_applied ?m)
               (configures ?m ?key)
               (configures ?m ?value))
)
