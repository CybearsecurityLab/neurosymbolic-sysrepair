```pddl
(:predicates
  (package_installed ?p - package)
  (package_unpacked ?p - package)
  (package_configured ?p - package)
  (package_removed ?p - package)
  (package_purged ?p - package)
  (triggers_pending ?p - package)
  (triggers_processed ?p - package)
  (package_verified ?p - package)
)

(:action install-package
  :parameters (?p - package ?file - filesystem_object)
  :precondition (and (not (package_installed ?p)) (file_exists ?file))
  :effect (and (package_installed ?p) (not (file_exists ?file)))
)

(:action unpack-package
  :parameters (?p - package ?file - filesystem_object)
  :precondition (and (not (package_installed ?p)) (not (package_unpacked ?p)) (file_exists ?file))
  :effect (and (package_unpacked ?p) (not (file_exists ?file)))
)

(:action configure-package
  :parameters (?p - package)
  :precondition (and (package_unpacked ?p) (not (package_configured ?p)))
  :effect (and (package_configured ?p) (not (package_unpacked ?p)))
)

(:action process-triggers
  :parameters (?p - package)
  :precondition (triggers_pending ?p)
  :effect (and (triggers_processed ?p) (not (triggers_pending ?p)))
)

(:action remove-package
  :parameters (?p - package)
  :precondition (package_installed ?p)
  :effect (and (package_removed ?p) (not (package_installed ?p)))
)

(:action purge-package
  :parameters (?p - package)
  :precondition (or (package_installed ?p) (package_removed ?p))
  :effect (and (package_purged ?p) (not (package_removed ?p)) (not (package_installed ?p)))
)

(:action verify-package
  :parameters (?p - package)
  :precondition (package_installed ?p)
  :effect (package_verified ?p)
)
```
