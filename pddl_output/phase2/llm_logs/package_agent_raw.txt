 :types
  package repository - object

:predicates
  (installed ?p - package)
  (available ?p - package)
  (in_repository ?p ?r - package repository)

:action install_package
  :parameters (?p - package ?r - repository)
  :precondition (and (available ?p) (not (installed ?p)) (in_repository ?p ?r))
  :effect (and (installed ?p))

:action uninstall_package
  :parameters (?p - package)
  :precondition (installed ?p)
  :effect (not (installed ?p))

:action update_repository
  :parameters (?r - repository)
  :precondition (in_repository ?p ?r)
  :effect (and (available ?p))

:action search_package
  :parameters (?name - string)
  :precondition (exists (?p - package) (has_name ?name ?p))
  :effect (and (in_repository ?p ?r))

:action add_to_repository
  :parameters (?p - package ?r - repository)
  :precondition (available ?p)
  :effect (and (in_repository ?p ?r))

:action remove_from_repository
  :parameters (?p - package ?r - repository)
  :precondition (in_repository ?p ?r)
  :effect (not (in_repository ?p ?r))

:action upgrade_package
  :parameters (?p - package)
  :precondition (installed ?p)
  :effect (and (available ?p) (uninstall_package ?p) (install_package ?p))

:action list_packages
  :parameters ()
  :precondition (true)
  :effect (exists (?p - package) (installed ?p))

:action list_repositories
  :parameters ()
  :precondition (true)
  :effect (exists (?r - repository))
