(:predicates 
  (acl_modified ?f - filesystem_object)
  (acl_removed ?f - filesystem_object)
  (acl_set ?f - filesystem_object)
  (acl_restored ?f - filesystem_object)
)

(:action modify_acl
  :parameters (?f - filesystem_object ?u - user ?g - group ?perms - string)
  :precondition (and (file_exists ?f) (can_escalate ?u))
  :effect (acl_modified ?f)
)

(:action remove_acl
  :parameters (?f - filesystem_object ?u - user ?g - group)
  :precondition (and (file_exists ?f) (acl_modified ?f))
  :effect (acl_removed ?f)
)

(:action set_acl
  :parameters (?f - filesystem_object ?u - user ?g - group ?perms - string)
  :precondition (file_exists ?f)
  :effect (acl_set ?f)
)

(:action restore_acl
  :parameters (?f - filesystem_object ?source - filesystem_object)
  :precondition (file_exists ?source)
  :effect (acl_restored ?f)
)
