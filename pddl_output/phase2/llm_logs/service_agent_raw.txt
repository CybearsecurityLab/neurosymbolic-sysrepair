 :types
  service - object
  process - object

:predicates
  (active ?s - service)
  (running ?p - process)
  (unit ?s - service)
  (process_of ?p - service)

:action list_units
  :parameters (?s - service)
  :precondition (true)
  :effect (and (active ?s))

:action systemctl_list_units
  :parameters (?s - service)
  :precondition (and (unit ?s))
  :effect (and (active ?s))

:action check_process
  :parameters (?p - process)
  :precondition (true)
  :effect (and (running ?p))

:action systemctl_check_process
  :parameters (?p - process)
  :precondition (and (process_of ?p))
  :effect (and (running ?p))
