```lisp
(:predicates
  (change_exists ?c - object)
  (change_aborted ?c - object)
  (assertion_valid ?a - object)
  (assertion_signed ?a - object)
  (assertion_added ?a - object)
  (snap_application_exists ?s - object)
  (alias_configured ?a - object ?s - object)
  (snapshot_exists ?sn - object)
  (user_data_exists ?u - user)
  (snapshot_verified ?sn - object)
  (plug_available ?p - object)
  (slot_available ?s - object)
  (plug_connected ?p - object ?s - object)
  (snap_installed ?s - object)
  (cohort_created ?c - object)
)

(:action abort-change
  :parameters (?c - object)
  :precondition (and (change_exists ?c))
  :effect (and (change_aborted ?c)
               (not (change_exists ?c)))
)

(:action add-assertion
  :parameters (?a - object)
  :precondition (and (assertion_valid ?a)
                     (assertion_signed ?a))
  :effect (assertion_added ?a)
)

(:action set-alias
  :parameters (?a - object ?s - object)
  :precondition (snap_application_exists ?s)
  :effect (alias_configured ?a ?s)
)

(:action verify-snapshot
  :parameters (?sn - object ?u - user)
  :precondition (and (snapshot_exists ?sn)
                     (user_data_exists ?u))
  :effect (snapshot_verified ?sn)
)

(:action connect-plug-slot
  :parameters (?p - object ?s - object)
  :precondition (and (plug_available ?p)
                     (slot_available ?s))
  :effect (plug_connected ?p ?s)
)

(:action generate-cohort
  :parameters (?s - object ?c - object)
  :precondition (snap_installed ?s)
  :effect (cohort_created ?c)
)
```
