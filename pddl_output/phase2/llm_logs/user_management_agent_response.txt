```pddl
(:types
  user group - object
)

(:predicates
  (user_exists ?u - user)
  (group_exists ?g - group)
  (member_of ?u - user ?g - group)
  (can_escalate ?u - user)
)

(:action create_user
  :parameters (?u - user ?g - group)
  :precondition (and (not (user_exists ?u)))
  :effect (and (user_exists ?u) (member_of ?u ?g))
)

(:action modify_user
  :parameters (?u - user ?g - group)
  :precondition (and (user_exists ?u))
  :effect (and (member_of ?u ?g))
)

(:action delete_user
  :parameters (?u - user)
  :precondition (and (user_exists ?u))
  :effect (not (user_exists ?u))
)

(:action create_group
  :parameters (?g - group)
  :precondition (and (not (group_exists ?g)))
  :effect (group_exists ?g)
)

(:action delete_group
  :parameters (?g - group)
  :precondition (and (group_exists ?g))
  :effect (not (group_exists ?g))
)
```
