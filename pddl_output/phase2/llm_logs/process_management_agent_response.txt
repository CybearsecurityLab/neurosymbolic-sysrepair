```pddl
(:types
  process - object
)

(:predicates
  (running ?p - process)
  (stopped ?p - process)
  (has_priority ?p - process ?n - number)
  (is_niced ?p - process)
  (is_nohungup ?p - process)
)

(:action start_process
  :parameters (?p - process)
  :precondition (not (running ?p))
  :effect (running ?p)
)

(:action stop_process
  :parameters (?p - process)
  :precondition (running ?p)
  :effect (stopped ?p)
)

(:action set_priority
  :parameters (?p - process ?n - number)
  :precondition (running ?p)
  :effect (has_priority ?p ?n)
)

(:action apply_nice
  :parameters (?p - process)
  :precondition (running ?p)
  :effect (is_niced ?p)
)

(:action apply_nohangup
  :parameters (?p - process)
  :precondition (running ?p)
  :effect (is_nohungup ?p)
)
```
