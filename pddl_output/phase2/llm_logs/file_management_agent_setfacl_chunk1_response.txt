```pddl
(:predicates
  (user_can_read ?u - user ?f - filesystem_object)
  (mask_has_permissions ?f - filesystem_object ?perms - object)
  (group_entry_exists ?g - group ?f - filesystem_object)
  (acl_copied ?dst - filesystem_object ?src - filesystem_object)
  (default_acl_set ?dir - directory)
)

(:action grant_user_read
  :parameters (?u - user ?f - filesystem_object)
  :precondition (file_exists ?f)
  :effect (user_can_read ?u ?f)
)

(:action set_mask_permissions
  :parameters (?f - filesystem_object ?perms - object)
  :precondition (file_exists ?f)
  :effect (mask_has_permissions ?f ?perms)
)

(:action remove_group_entry
  :parameters (?g - group ?f - filesystem_object)
  :precondition (and (group_exists ?g) (file_exists ?f))
  :effect (not (group_entry_exists ?g ?f))
)

(:action copy_acl
  :parameters (?src - filesystem_object ?dst - filesystem_object)
  :precondition (and (file_exists ?src) (file_exists ?dst))
  :effect (acl_copied ?dst ?src)
)

(:action set_default_acl
  :parameters (?dir - directory)
  :precondition (file_exists ?dir)
  :effect (default_acl_set ?dir)
)
```
