```pddl
(define (problem sudo_actions)
  (:domain sudo)
  (:objects
    user1 user2 - user
    process1 process2 - process
    shell_path chroot_dir - filesystem_object
    role1 role2 - role
    type1 type2 - type
    timeout1 timeout2 - timeout
  )
  (:init
    (user_exists user1)
    (user_exists user2)
    (file_exists shell_path)
    (file_exists chroot_dir)
    (cached_credentials_valid user1)
    (cached_credentials_valid user2)
    (can_escalate user1)
    (can_escalate user2)
  )
  (:goal
    (and
      (not (cached_credentials_valid user1))
      (can_escalate user2)
      (executed_as_user process1 user1)
      (chroot_set process2 chroot_dir)
      (selinux_role_set process1 role1)
      (selinux_type_set process1 type1)
      (command_timeout_set process1 timeout1)
    )
  )
)
```

```pddl
(define (domain sudo)
  (:requirements :strips :typing)
  (:types
    user process filesystem_object role type timeout
  )
  (:predicates
    (user_exists ?u - user)
    (file_exists ?f - filesystem_object)
    (cached_credentials_valid ?u - user)
    (can_escalate ?u - user)
    (executed_as_user ?p - process ?u - user)
    (chroot_set ?p - process ?d - filesystem_object)
    (selinux_role_set ?p - process ?r - role)
    (selinux_type_set ?p - process ?t - type)
    (command_timeout_set ?p - process ?t - timeout)
    (interactive_prompt_required ?p - process)
    (groups_preserved ?p - process)
  )
  (:action sudo_remove_cached_credentials
    :parameters (?u - user)
    :precondition (and (user_exists ?u) (cached_credentials_valid ?u))
    :effect (not (cached_credentials_valid ?u))
  )
  (:action sudo_reset_timestamp
    :parameters (?u - user)
    :precondition (and (user_exists ?u) (cached_credentials_valid ?u))
    :effect (not (cached_credentials_valid ?u))
  )
  (:action sudo_list_privileges
    :parameters (?u - user)
    :precondition (user_exists ?u)
    :effect (can_escalate ?u)
  )
  (:action sudo_run_login_shell
    :parameters (?u - user ?p - process)
    :precondition (and (user_exists ?u) (file_exists shell_path))
    :effect (and (executed_as_user ?p ?u) (groups_preserved ?p))
  )
  (:action sudo_chroot
    :parameters (?p - process ?d - filesystem_object)
    :precondition (file_exists ?d)
    :effect (chroot_set ?p ?d)
  )
  (:action sudo_set_selinux_role
    :parameters (?p - process ?r - role)
    :precondition (process_exists ?p)
    :effect (selinux_role_set ?p ?r)
  )
  (:action sudo_set_selinux_type
    :parameters (?p - process ?t - type)
    :precondition (process_exists ?p)
    :effect (selinux_type_set ?p ?t)
  )
  (:action sudo_set_timeout
    :parameters (?p - process ?t - timeout)
    :precondition (process_exists ?p)
    :effect (command_timeout_set ?p ?t)
  )
  (:action sudo_disable_interactive_prompt
    :parameters (?p - process)
    :precondition (process_exists ?p)
    :effect (not (interactive_prompt_required ?p))
  )
  (:action sudo_preserve_groups
    :parameters (?p - process)
    :precondition (process_exists ?p)
    :effect (groups_preserved ?p)
  )
)
```
