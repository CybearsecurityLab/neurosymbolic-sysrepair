```pddl
(define (problem disable-snap-problem)
  (:domain snap-management)
  (:objects 
    snap1 - snap
    plug1 slot1 - plug slot
    file1 - filesystem_object
    change1 - change
    channel1 - channel
    revision1 - revision
  )
  (:init 
    (snap_enabled snap1)
    (plug_connected plug1 slot1)
    (file_exists file1)
    (change_pending change1)
  )
  (:goal 
    (and 
      (snap_disabled snap1)
      (plug_disconnected plug1 slot1)
      (snapshot_imported file1)
      (change_aborted change1)
    )
  )
)
```

```pddl
(define (domain snap-management)
  (:requirements :strips :typing)
  (:types 
    snap plug slot filesystem_object change channel revision
  )
  (:predicates 
    (snap_enabled ?s - snap)
    (snap_disabled ?s - snap)
    (plug_connected ?p - plug ?s - slot)
    (plug_disconnected ?p - plug ?s - slot)
    (connection_forget ?p - plug ?s - slot)
    (file_exists ?f - filesystem_object)
    (snapshot_imported ?f - filesystem_object)
    (change_pending ?c - change)
    (change_aborted ?c - change)
  )
  (:action disable_snap
    :parameters (?s - snap)
    :precondition (snap_enabled ?s)
    :effect (and 
      (not (snap_enabled ?s)) 
      (snap_disabled ?s)
    )
  )
  (:action enable_snap
    :parameters (?s - snap)
    :precondition (snap_disabled ?s)
    :effect (and 
      (not (snap_disabled ?s)) 
      (snap_enabled ?s)
    )
  )
  (:action disconnect_plug_slot
    :parameters (?p - plug ?s - slot ?forget - boolean)
    :precondition (plug_connected ?p ?s)
    :effect (and 
      (not (plug_connected ?p ?s)) 
      (plug_disconnected ?p ?s)
      (when ?forget (connection_forget ?p ?s))
    )
  )
  (:action download_snap
    :parameters (?s - snap ?channel - channel ?revision - revision)
    :precondition (and)
    :effect (file_exists ?f)
  )
  (:action import_snapshot
    :parameters (?f - filesystem_object)
    :precondition (file_exists ?f)
    :effect (snapshot_imported ?f)
  )
  (:action abort_change
    :parameters (?c - change)
    :precondition (change_pending ?c)
    :effect (change_aborted ?c)
  )
)
```
