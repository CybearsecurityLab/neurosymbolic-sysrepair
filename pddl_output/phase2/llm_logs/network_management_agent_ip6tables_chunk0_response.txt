```lisp
(:predicates
  (firewall_rule_exists ?r - firewall_rule)
)

(:action append-firewall-rule
  :parameters (?chain - firewall_rule ?rule - firewall_rule)
  :precondition (and (firewall_rule_exists ?chain) (not (firewall_rule_exists ?rule)))
  :effect (firewall_rule_exists ?rule)
)

(:action delete-firewall-rule
  :parameters (?chain - firewall_rule ?rule - firewall_rule)
  :precondition (firewall_rule_exists ?rule)
  :effect (not (firewall_rule_exists ?rule))
)

(:action insert-firewall-rule
  :parameters (?chain - firewall_rule ?rule - firewall_rule)
  :precondition (and (firewall_rule_exists ?chain) (not (firewall_rule_exists ?rule)))
  :effect (firewall_rule_exists ?rule)
)

(:action replace-firewall-rule
  :parameters (?chain - firewall_rule ?old_rule - firewall_rule ?new_rule - firewall_rule)
  :precondition (and (firewall_rule_exists ?old_rule) (not (firewall_rule_exists ?new_rule)))
  :effect (and (not (firewall_rule_exists ?old_rule)) (firewall_rule_exists ?new_rule))
)

(:action flush-firewall-chain
  :parameters (?chain - firewall_rule)
  :precondition (firewall_rule_exists ?chain)
  :effect (forall (?r - firewall_rule) (when (firewall_rule_exists ?r) (not (firewall_rule_exists ?r))))
)

(:action create-firewall-chain
  :parameters (?chain - firewall_rule)
  :precondition (not (firewall_rule_exists ?chain))
  :effect (firewall_rule_exists ?chain)
)

(:action delete-firewall-chain
  :parameters (?chain - firewall_rule)
  :precondition (firewall_rule_exists ?chain)
  :effect (not (firewall_rule_exists ?chain))
)
```
