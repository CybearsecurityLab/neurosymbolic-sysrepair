 :types
  file - object
  directory - object

(:predicates)
(file_exists ?f - file)
(directory_exists ?d - directory)
(file_in_directory ?f - file ?d - directory)
(file_is_directory ?f - file)
(directory_is_empty ?d - directory)
(file_is_readable ?f - file)
(file_is_writable ?f - file)
(file_is_executable ?f - file)
(file_is_owned_by ?f - file ?u - user)
(file_is_group_owned_by ?f - file ?g - group)
(file_has_permissions ?f - file (read write execute))
(directory_has_permissions ?d - directory (read write execute))

(:action copy_file
  :parameters (?src_file - file ?dest_file - file)
  :precondition (and (file_exists ?src_file) (directory_exists (parent ?dest_file)))
  :effect (and (not (file_exists ?dest_file)) (file_exists ?dest_file) (file_in_directory ?dest_file ?src_file) (not (file_is_same_as ?src_file ?dest_file))))

(:action move_file
  :parameters (?src_file - file ?dest_dir - directory)
  :precondition (and (file_exists ?src_file) (directory_exists ?dest_dir))
  :effect (and (not (file_exists ?src_file)) (file_exists ?dest_file) (file_in_directory ?dest_dir ?src_file) (not (file_is_same_as ?src_file ?dest_file))))

(:action rename_file
  :parameters (?src_file - file ?new_name - string)
  :precondition (and (file_exists ?src_file) (not (file_exists (rename ?src_file ?new_name))))
  :effect (and (not (file_exists ?src_file)) (file_exists (rename ?src_file ?new_name)) (file_in_directory (parent (rename ?src_file ?new_name)) ?src_file) (not (file_is_same_as ?src_file (rename ?src_file ?new_name)))))

(:action create_directory
  :parameters (?dir - directory)
  :precondition (not (directory_exists ?dir)))
  :effect (directory_exists ?dir))

(:action delete_directory
  :parameters (?dir - directory)
  :precondition (directory_exists ?dir)
  :effect (and (not (directory_exists ?dir)) (forall (?file - file) (implies (file_in_directory ?dir ?file) (not (file_exists ?file))))))

(:action delete_file
  :parameters (?file - file)
  :precondition (file_exists ?file)
  :effect (not (file_exists ?file)))

(:action change_permissions_file
  :parameters (?file - file ?permissions - (read write execute))
  :precondition (file_exists ?file))
  :effect (file_has_permissions ?file ?permissions))

(:action change_permissions_directory
  :parameters (?dir - directory ?permissions - (read write execute))
  :precondition (directory_exists ?dir))
  :effect (directory_has_permissions ?dir ?permissions))

(:action change_owner_file
  :parameters (?file - file ?new_owner - user))
  :precondition (file_exists ?file))
  :effect (file_is_owned_by ?file ?new_owner))

(:action change_owner_directory
  :parameters (?dir - directory ?new_owner - user))
  :precondition (directory_exists ?dir))
  :effect (directory_is_owned_by ?dir ?new_owner))

(:action change_group_file
  :parameters (?file - file ?new_group - group))
  :precondition (file_exists ?file))
  :effect (file_is_group_owned_by ?file ?new_group))

(:action change_group_directory
  :parameters (?dir - directory ?new_group - group))
  :precondition (directory_exists ?dir))
  :effect (directory_is_group_owned_by ?dir ?new_group))
