(:types
  file directory - object
)

(:predicates
  (file_exists ?f - file)
  (directory_exists ?d - directory)
  (file_in_directory ?f - file ?d - directory)
  (directory_empty ?d - directory)
  (file_size ?f - file ?size - integer)
  (file_owner ?f - file ?u - user)
  (file_group ?f - file ?g - group)
  (file_permissions ?f - file ?perm - string)
  (file_symlink_target ?f - file ?target - file)
)

(:action create_file
  :parameters (?f - file)
  :precondition (not (file_exists ?f))
  :effect (file_exists ?f)
)

(:action delete_file
  :parameters (?f - file)
  :precondition (file_exists ?f)
  :effect (not (file_exists ?f))
)

(:action rename_file
  :parameters (?old_f - file ?new_f - file)
  :precondition (and (file_exists ?old_f) (not (file_exists ?new_f)))
  :effect (and (not (file_exists ?old_f)) (file_exists ?new_f))
)

(:action create_directory
  :parameters (?d - directory)
  :precondition (not (directory_exists ?d))
  :effect (directory_exists ?d)
)

(:action delete_directory
  :parameters (?d - directory)
  :precondition (and (directory_exists ?d) (directory_empty ?d))
  :effect (not (directory_exists ?d))
)

(:action rename_directory
  :parameters (?old_d - directory ?new_d - directory)
  :precondition (and (directory_exists ?old_d) (not (directory_exists ?new_d)) (directory_empty ?old_d))
  :effect (and (not (directory_exists ?old_d)) (directory_exists ?new_d))
)

(:action add_file_to_directory
  :parameters (?f - file ?d - directory)
  :precondition (and (file_exists ?f) (not (file_in_directory ?f ?d)))
  :effect (file_in_directory ?f ?d)
)

(:action remove_file_from_directory
  :parameters (?f - file ?d - directory)
  :precondition (and (file_exists ?f) (file_in_directory ?f ?d))
  :effect (not (file_in_directory ?f ?d))
)

(:action set_file_owner
  :parameters (?f - file ?u - user)
  :precondition (file_exists ?f)
  :effect (file_owner ?f ?u)
)

(:action set_file_group
  :parameters (?f - file ?g - group)
  :precondition (file_exists ?f)
  :effect (file_group ?f ?g)
)

(:action set_file_permissions
  :parameters (?f - file ?perm - string)
  :precondition (file_exists ?f)
  :effect (file_permissions ?f ?perm)
)

(:action create_symlink
  :parameters (?symlink - file ?target - file)
  :precondition (and (file_exists ?target) (not (file_exists ?symlink)))
  :effect (and (file_exists ?symlink) (file_symlink_target ?symlink ?target))
)

(:action delete_symlink
  :parameters (?symlink - file)
  :precondition (and (file_exists ?symlink) (file_symlink_target ?symlink ?target))
  :effect (not (file_exists ?symlink))
)
