```lisp
(:predicates
  (system_change_exists ?c - object)
  (change_pending ?c - object)
  (change_aborted ?c - object)
  (assertion_valid ?a - object)
  (assertion_added ?a - object)
  (plug_exists ?p - object)
  (slot_exists ?s - object)
  (connected ?p - object ?s - object)
  (snapshot_exists ?sn - object)
  (snapshot_checked ?sn - object)
  (cohort_created ?c - object)
  (snap_in_cohort ?s - object ?c - object))

(:action abort_change
 :parameters (?c - object)
 :precondition (and (system_change_exists ?c) (change_pending ?c))
 :effect (and (not (change_pending ?c)) (change_aborted ?c)))

(:action add_assertion
 :parameters (?a - object)
 :precondition (assertion_valid ?a)
 :effect (assertion_added ?a))

(:action connect_plug_slot
 :parameters (?p - object ?s - object)
 :precondition (and (plug_exists ?p) (slot_exists ?s))
 :effect (connected ?p ?s))

(:action check_snapshot
 :parameters (?sn - object)
 :precondition (snapshot_exists ?sn)
 :effect (snapshot_checked ?sn))

(:action create_cohort
 :parameters (?c - object ?s - object)
 :precondition (and (cohort_created ?c) (snap_in_cohort ?s ?c))
 :effect (and (not (cohort_created ?c)) (snap_in_cohort ?s ?c)))
```
